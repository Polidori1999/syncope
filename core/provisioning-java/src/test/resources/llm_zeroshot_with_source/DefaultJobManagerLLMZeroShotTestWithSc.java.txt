    @Before
    public void setUp() {
        MockitoAnnotations.initMocks(this);

        when(ctx.getBeanFactory()).thenReturn(beanFactory);
        // Default behavior: createBean returns a mock Job instance for any requested class
        when(beanFactory.createBean(any(Class.class))).thenAnswer(invocation -> {
            Class<?> cls = invocation.getArgument(0);
            // create a Mockito mock for the Job type
            Object jobMock = mock(Job.class);
            return jobMock;
        });

        when(securityProperties.getAdminUser()).thenReturn("admin");

        jobManager = new DefaultJobManager(
                domainHolder,

                implementationDAO,
                taskUtilsFactory,
                confParamOps,
                securityProperties,
                ctx);

    /**
     * Test isRunning behavior:
     * - when jobStatusDAO.lock returns true, isRunning should return false (not running) and unlock should be called.
     * - when jobStatusDAO.lock returns false, isRunning should return true (running) and unlock should not be called.
     */
    @Test
    public void testIsRunningLockingBehavior() {
        String jobName = "jobA";

        // Case 1: lock returns true -> not running
        when(jobStatusDAO.lock(jobName)).thenReturn(true);
        boolean running = jobManager.isRunning(jobName);
        assertFalse("Expected not running when lock returns true", running);
        verify(jobStatusDAO).unlock(jobName);

        // Case 2: lock returns false -> running
        reset(jobStatusDAO);
        when(jobStatusDAO.lock(jobName)).thenReturn(false);
        running = jobManager.isRunning(jobName);
        assertTrue("Expected running when lock returns false", running);
        verify(jobStatusDAO, never()).unlock(jobName);
    }

    /**
     * Test registerJob schedules a job via scheduler.register when no cronExpression and no startAt are provided.
     */
    @Test
    public void testRegisterJobSchedulesRegister() {
        // Prepare context
        JobExecutionContext context = new JobExecutionContext("domain", "jobName", "executor", false);

        // Ensure isRunning returns false (so registerJob proceeds)
        when(jobStatusDAO.lock(context.getJobName())).thenReturn(true);

        // Call protected method via reflection
        jobManager.registerJob(context, TaskJob.class, null, null);

        // verify scheduler.register called with a Job instance
        verify(scheduler).register(any(Job.class));
    }

    /**
     * Test registerJob schedules with CronTrigger when cronExpression provided and not running.
     */
    @Test
    public void testRegisterJobSchedulesCronTrigger() {
        JobExecutionContext context = new JobExecutionContext("domain", "jobCron", "executor", false);
        when(jobStatusDAO.lock(context.getJobName())).thenReturn(true);

        String cron = "0 0 12 * * ?";
        jobManager.registerJob(context, TaskJob.class, cron, null);

        // verify scheduler.schedule called with CronTrigger
        ArgumentCaptor<Object> captor = ArgumentCaptor.forClass(Object.class);
        verify(scheduler).schedule(any(Job.class), any(CronTrigger.class));
    }

    /**
     * Test register for SchedTask:
     * - active PullTask with no explicit delegate should resolve PullJobDelegate implementation and schedule.
     * - inactive task should be skipped.
     * - missing delegate should throw IllegalArgumentException.
     */
    @Test
    public void testRegisterSchedTaskVariousPaths() {
        // Prepare a PullTask mock (SchedTask)
        PullTask pullTask = mock(PullTask.class);
        when(pullTask.isActive()).thenReturn(true);
        when(pullTask.getKey()).thenReturn("pullKey");
        when(pullTask.getName()).thenReturn("pullName");
        when(pullTask.getCronExpression()).thenReturn("0 0 1 * * ?");
        when(taskUtilsFactory.getInstance(pullTask)).thenReturn(new org.apache.syncope.core.persistence.api.entity.task.TaskUtils() {
            @Override
            public TaskType getType() {
                return TaskType.PULL;
            }
        });

        // Provide an Implementation that matches PullJobDelegate body
        Implementation impl = mock(Implementation.class);
        when(impl.getBody()).thenReturn(PullJobDelegate.class.getName());
        when(impl.getKey()).thenReturn("implKey");
        when(implementationDAO.findByType(IdRepoImplementationType.TASKJOB_DELEGATE)).thenReturn(List.of(impl));

        // jobDelegate is null on task, so the code should pick the implementation above
        when(jobStatusDAO.lock(anyString())).thenReturn(true);

        // Call protected register via public execute wrapper
        jobManager.execute(pullTask, OffsetDateTime.now(), "executor", false, Map.of());

        // verify scheduler.register or schedule invoked (cronExpression null in execute -> startAt provided)
        // In execute we pass startAt, so register() will call register with startAt not null; ensure schedule called
        verify(scheduler, atLeastOnce()).schedule(any(Job.class), any());
    }

    @Test
    public void testRegisterInactiveTaskSkippedAndMissingDelegateThrows() {
        // Inactive task should be skipped
        SchedTask inactive = mock(SchedTask.class);
        when(inactive.isActive()).thenReturn(false);
        when(taskUtilsFactory.getInstance(inactive)).thenReturn(new org.apache.syncope.core.persistence.api.entity.task.TaskUtils() {
            @Override
            public TaskType getType() {
                return TaskType.SCHEDULED;
            }
        });
        // Should not throw
        jobManager.register(AuthContextUtils.getDomain(), inactive, null, null, "exec", false, Map.of());
        // No interactions with scheduler
        verifyNoInteractions(scheduler);

        // Active task but no delegate and no matching implementation -> expect IllegalArgumentException
        SchedTask activeNoDelegate = mock(SchedTask.class);
        when(activeNoDelegate.isActive()).thenReturn(true);
        when(activeNoDelegate.getKey()).thenReturn("k1");
        when(activeNoDelegate.getName()).thenReturn("n1");
        when(taskUtilsFactory.getInstance(activeNoDelegate)).thenReturn(new org.apache.syncope.core.persistence.api.entity.task.TaskUtils() {
            @Override
            public TaskType getType() {
                return TaskType.SCHEDULED;
            }
        });
        when(implementationDAO.findByType(IdRepoImplementationType.TASKJOB_DELEGATE)).thenReturn(List.of());
        try {
            jobManager.register(AuthContextUtils.getDomain(), activeNoDelegate, null, null, "exec", false, Map.of());
            fail("Expected IllegalArgumentException when no delegate implementation is available");
        } catch (IllegalArgumentException e) {
            // expected
        }
    }

    /**
     * Test unregister wrappers for Task and Report call unregisterJob with proper job names.
     */
    @Test
    public void testUnregisterWrappers() {
        Task<?> task = mock(Task.class);
        when(task.getKey()).thenReturn("tk");
        // JobNamer.getJobName uses task; we cannot easily predict its format, but we can verify scheduler interactions indirectly
        // Provide a simple SchedTask to unregister
        SchedTask st = mock(SchedTask.class);
        when(st.getKey()).thenReturn("tsk");
        when(st.getName()).thenReturn("tname");
        jobManager.unregister(st);
        // Since unregisterJob uses AuthContextUtils.getDomain() inside, verify scheduler stop/delete called
        verify(scheduler).stop(anyString(), anyString());
        verify(scheduler).delete(anyString(), anyString());

        Report report = mock(Report.class);
        when(report.getKey()).thenReturn("rk");
        jobManager.unregister(report);
        verify(scheduler, atLeast(2)).stop(anyString(), anyString());
        verify(scheduler, atLeast(2)).delete(anyString(), anyString());
    }

    /**
     * Test load and unload with empty DAOs do not throw and interact with scheduler as expected.

    @Test
    public void testLoadAndUnloadNoItems() {
        // No tasks or reports
        when(taskDAO.findAll(TaskType.SCHEDULED)).thenReturn(List.of());
        when(taskDAO.findAll(TaskType.PULL)).thenReturn(List.of());
        when(taskDAO.findAll(TaskType.PUSH)).thenReturn(List.of());
        when(taskDAO.findAll(TaskType.MACRO)).thenReturn(List.of());
        when(taskDAO.findAll(TaskType.LIVE_SYNC)).thenReturn(List.of());
        when(reportDAO.findAll()).thenReturn(List.of());

        // Call load for a non-master domain to avoid NotificationJob scheduling
        jobManager.load("someDomain");
        // No exceptions and no scheduling for notification/system reporter
        verify(beanFactory, never()).createBean(NotificationJob.class);
        verify(beanFactory, never()).createBean(SystemLoadReporterJob.class);

        // Call unload - should attempt to stop/delete nothing but must not throw
        jobManager.unload("someDomain");
    }

    /**
     * Test load for MASTER_DOMAIN triggers notification and system reporter scheduling when confParamOps returns values.
     */
    @Test
    public void testLoadMasterDomainSchedulesNotificationAndSystemReporter() {
        // Prepare empty tasks and reports
        when(taskDAO.findAll(TaskType.SCHEDULED)).thenReturn(List.of());
        when(taskDAO.findAll(TaskType.PULL)).thenReturn(List.of());
        when(taskDAO.findAll(TaskType.PUSH)).thenReturn(List.of());
        when(taskDAO.findAll(TaskType.MACRO)).thenReturn(List.of());
        when(taskDAO.findAll(TaskType.LIVE_SYNC)).thenReturn(List.of());
        when(reportDAO.findAll()).thenReturn(List.of());

        // confParamOps returns null -> should use NotificationJob.DEFAULT_CRON_EXP
        when(confParamOps.get(eq(SyncopeConstants.MASTER_DOMAIN), eq("notificationjob.cronExpression"), any(), eq(String.class)))
                .thenReturn(null);

        // Ensure lock returns true so registerJob proceeds
        when(jobStatusDAO.lock(anyString())).thenReturn(true);

        // Call load for MASTER_DOMAIN
        jobManager.load(SyncopeConstants.MASTER_DOMAIN);

        // NotificationJob and SystemLoadReporterJob should be scheduled via beanFactory.createBean calls
        verify(beanFactory, atLeastOnce()).createBean(NotificationJob.class);
        verify(beanFactory, atLeastOnce()).createBean(SystemLoadReporterJob.class);
    }
