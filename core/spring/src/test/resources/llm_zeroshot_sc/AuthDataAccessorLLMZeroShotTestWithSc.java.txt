    // Test authenticate (username/password) returns null when no user found
    @Test
    public void testAuthenticateNoUserFound() {

        Authentication authentication = mock(Authentication.class);
        when(authentication.getName()).thenReturn("unknown");

    // Test authenticate increments failed logins on bad password and saves user
    @Test
    public void testAuthenticateBadPasswordIncrementsFailedLogins() {
        // Short comment: failed authentication should increment failedLogins and persist user
        Authentication authentication = mock(Authentication.class);
        when(authentication.getName()).thenReturn("user1");
        when(authentication.getCredentials()).thenReturn("badpass");
        SyncopeAuthenticationDetails details = mock(SyncopeAuthenticationDetails.class);
        when(authentication.getDetails()).thenReturn(details);

        when(confParamOps.get(eq("domain"), eq("authentication.attributes"), any(), eq(String[].class)))
                .thenReturn(new String[] { "username" });
        when(confParamOps.get(eq("domain"), eq("authentication.statuses"), any(), eq(String[].class)))
                .thenReturn(new String[] {});
        User user = mock(User.class);
        when(userDAO.findByUsername("user1")).thenReturn(Optional.of(user));
        when(user.isSuspended()).thenReturn(Boolean.FALSE);
        when(user.getStatus()).thenReturn(null);
        when(user.getFailedLogins()).thenReturn(0);
        when(encryptorManager.getInstance()).thenReturn(mock(EncryptorManager.Encryptor.class));
        when(encryptorManager.getInstance().verify(anyString(), anyString(), anyString())).thenReturn(false);
        when(userDAO.save(user)).thenReturn(user);

        AuthDataAccessor.UsernamePasswordAuthResult result =
                authDataAccessor.authenticate("domain", authentication);

        assertNotNull(result.user);
        assertFalse(result.authenticated);
    }

    // Test authenticate successful resets failed logins and sets last login date when configured
    @Test
    public void testAuthenticateSuccessResetsFailedLoginsAndSetsLastLogin() {
        // Short comment: successful authentication resets failed logins and updates last login date
        Authentication authentication = mock(Authentication.class);
        when(authentication.getName()).thenReturn("user2");
        when(authentication.getCredentials()).thenReturn("goodpass");
        SyncopeAuthenticationDetails details = mock(SyncopeAuthenticationDetails.class);
        when(authentication.getDetails()).thenReturn(details);
        when(details.getDelegatedBy()).thenReturn(null);

        when(confParamOps.get(eq("domain"), eq("authentication.attributes"), any(), eq(String[].class)))
                .thenReturn(new String[] { "username" });
        when(confParamOps.get(eq("domain"), eq("authentication.statuses"), any(), eq(String[].class)))
                .thenReturn(new String[] {});
        when(confParamOps.get(eq("domain"), eq("log.lastlogindate"), eq(true), eq(Boolean.class)))
                .thenReturn(true);

        User user = mock(User.class);
        when(userDAO.findByUsername("user2")).thenReturn(Optional.of(user));
        when(user.isSuspended()).thenReturn(Boolean.FALSE);
        when(user.getStatus()).thenReturn(null);
        when(user.getFailedLogins()).thenReturn(2);
        when(user.getKey()).thenReturn("user2Key");

        EncryptorManager.Encryptor encryptor = mock(EncryptorManager.Encryptor.class);
        when(encryptorManager.getInstance()).thenReturn(encryptor);
        when(encryptor.verify("goodpass", user.getCipherAlgorithm(), user.getPassword())).thenReturn(true);

        when(userDAO.save(user)).thenReturn(user);

        AuthDataAccessor.UsernamePasswordAuthResult result =
                authDataAccessor.authenticate("domain", authentication);

        assertNotNull(result.user);
        assertTrue(result.authenticated);
        // delegationKey should be null because details.getDelegatedBy() returns null
        assertNull(result.delegationKey);
    }

    // Test usernamePasswordAuthentication uses passthrough resources when internal verify fails
    @Test
    public void testUsernamePasswordAuthenticationWithPassthrough() throws Exception {
        // Short comment: if internal verify fails, passthrough resources are tried
        User user = mock(User.class);
        when(user.getUsername()).thenReturn("passthroughUser");
        when(user.getCipherAlgorithm()).thenReturn("alg");
        when(user.getPassword()).thenReturn("pw");

        EncryptorManager.Encryptor encryptor = mock(EncryptorManager.Encryptor.class);
        when(encryptorManager.getInstance()).thenReturn(encryptor);
        when(encryptor.verify(anyString(), anyString(), anyString())).thenReturn(false);

        // create a fake ExternalResource and Provision
        ExternalResource resource = mock(ExternalResource.class);
        when(resource.getKey()).thenReturn("res1");
        Provision provision = mock(Provision.class);
        when(resource.getProvisionByAnyType("USER")).thenReturn(Optional.of(provision));
        when(mappingManager.getConnObjectKeyValue(eq(user), eq(resource), eq(provision))).thenReturn(Optional.of("connKey"));

        // connector authenticate returns a non-null Uid
        org.identityconnectors.framework.api.ConnectorFacade connector = mock(org.identityconnectors.framework.api.ConnectorFacade.class);
        when(connectorManager.getConnector(resource)).thenReturn(connector);
        when(connector.authenticate("connKey", "secret", null)).thenReturn(new Uid("uid"));

        // getPassthroughResources should return the resource
        when(userDAO.findAllResources(user)).thenReturn(Set.of(resource));
        when(resourceDAO.findByPolicy(any())).thenReturn(Set.of(resource));
        when(realmSearchDAO.findAncestors(any())).thenReturn(List.of());

        boolean authenticated = authDataAccessor.usernamePasswordAuthentication(user, "secret");
        assertTrue(authenticated);
    }

    // Test getPassthroughResources returns resources from both user and realm policies
    @Test
    public void testGetPassthroughResourcesCombinesUserAndRealm() {
        // Short comment: resources from user and realm account policies are combined/intersected correctly
        User user = mock(User.class);
        Object accountPolicy = new Object();

        ExternalResource r1 = mock(ExternalResource.class);
        ExternalResource r2 = mock(ExternalResource.class);

        when(userDAO.findAllResources(user)).thenReturn(Set.of(r1));
        when(r1.getAccountPolicy()).thenReturn(accountPolicy);
        when(resourceDAO.findByPolicy(accountPolicy)).thenReturn(Set.of(r1, r2));

        Realm realm = mock(Realm.class);
        when(realm.getAccountPolicy()).thenReturn(accountPolicy);
        when(user.getRealm()).thenReturn(realm);
        when(realmSearchDAO.findAncestors(realm)).thenReturn(List.of(realm));

        Set<ExternalResource> result = authDataAccessor.getPassthroughResources(user);
        assertNotNull(result);
        assertTrue(result.contains(r1));
        assertTrue(result.contains(r2));
    }

    // Test buildAuthorities creates SyncopeGrantedAuthority entries
    @Test
    public void testBuildAuthorities() {
        // Short comment: buildAuthorities should convert entitlement->realms map into authorities
        Map<String, Set<String>> entForRealms = new HashMap<>();
        entForRealms.put("ENT_A", Set.of("/realm"));
        Set<SyncopeGrantedAuthority> authorities = authDataAccessor.buildAuthorities(entForRealms);
        assertNotNull(authorities);
        assertFalse(authorities.isEmpty());
        boolean found = authorities.stream().anyMatch(a -> "ENT_A".equals(a.getAuthority()));
        assertTrue(found);
    }

    // Test authenticate(JWT) resolves provider and throws when resolved claims missing
    @Test(expected = AuthenticationCredentialsNotFoundException.class)
    public void testAuthenticateJWTProviderResolveMissing() {
        // Short comment: JWT with issuer not resolvable should raise AuthenticationCredentialsNotFoundException
        JWTAuthentication jwtAuth = mock(JWTAuthentication.class);
        JWTClaims claims = mock(JWTClaims.class);
        when(jwtAuth.getClaims()).thenReturn(claims);
        when(claims.getSubject()).thenReturn("someUser");
        when(claims.getIssuer()).thenReturn("issuer1");
        when(claims.getJWTID()).thenReturn("jwtid2");

        JWTSSOProvider provider = mock(JWTSSOProvider.class);
        when(jwtSSOProviders.stream()).thenReturn(List.of(provider).stream());
        when(provider.getIssuer()).thenReturn("issuer1");
        when(provider.resolve(claims)).thenReturn(Optional.empty());

        authDataAccessor = new AuthDataAccessor(securityProperties, encryptorManager, realmSearchDAO, userDAO,
                groupDAO, anySearchDAO, accessTokenDAO, confParamOps, roleDAO, delegationDAO, resourceDAO,
                connectorManager, auditManager, mappingManager, List.of(provider));

        authDataAccessor.authenticate(jwtAuth);
    }

    // Test authenticate(JWT) successful path with resolved claims and mustChangePassword resets authorities
    @Test
    public void testAuthenticateJWTResolvedMustChangePassword() {
        // Short comment: when resolved user must change password, authorities are reset to MUST_CHANGE_PASSWORD_AUTHORITIES
        JWTAuthentication jwtAuth = mock(JWTAuthentication.class);
        JWTClaims claims = mock(JWTClaims.class);
        when(jwtAuth.getClaims()).thenReturn(claims);
        when(claims.getSubject()).thenReturn("someUser");
        when(claims.getIssuer()).thenReturn("issuer2");
        when(claims.getJWTID()).thenReturn("jwtid3");

        User user = mock(User.class);
        when(user.isSuspended()).thenReturn(Boolean.FALSE);
        when(user.isMustChangePassword()).thenReturn(Boolean.TRUE);
        when(user.getStatus()).thenReturn("ACTIVE");
        when(user.getKey()).thenReturn("uKey");
        when(user.getUsername()).thenReturn("someUser");

        JWTSSOProvider provider = mock(JWTSSOProvider.class);
        JWTSSOProvider.ResolvedClaims resolved = mock(JWTSSOProvider.ResolvedClaims.class);
        when(resolved.user()).thenReturn(user);
        when(resolved.authorities()).thenReturn(Set.of(new SyncopeGrantedAuthority("X")));
        when(provider.getIssuer()).thenReturn("issuer2");
        when(provider.resolve(claims)).thenReturn(Optional.of(resolved));

        when(jwtSSOProviders.stream()).thenReturn(List.of(provider).stream());

        authDataAccessor = new AuthDataAccessor(securityProperties, encryptorManager, realmSearchDAO, userDAO,
                groupDAO, anySearchDAO, accessTokenDAO, confParamOps, roleDAO, delegationDAO, resourceDAO,
                connectorManager, auditManager, mappingManager, List.of(provider));

        SyncopeAuthenticationDetails details = mock(SyncopeAuthenticationDetails.class);
        when(details.getDomain()).thenReturn("domain");
        when(jwtAuth.getDetails()).thenReturn(details);

        when(confParamOps.get("domain", "authentication.statuses", new String[] {}, String[].class))
                .thenReturn(new String[] { "ACTIVE" });

        AuthDataAccessor.JWTAuthResult result = authDataAccessor.authenticate(jwtAuth);
        assertEquals("someUser", result.username);
        assertNotNull(result.authorities);
        // must change password authorities should be returned
        assertTrue(result.authorities.stream().anyMatch(a -> a.getAuthority().contains(IdRepoEntitlement.MUST_CHANGE_PASSWORD)));
    }
